刚体定轴转动的转动定律

绕定轴转动的刚体的角速度 {
  与作用于刚体上的合外力 {
    成正比
  }

  与刚体的转动惯量 {
    成反比
  }
}

转动惯量

J = {
  单个质点绕某轴运动 {
    m r(2)
  }

  分立质点绕同一轴转动 {
    sum[i] {
      Delta m[i] r[i](2)
    }
  }

  质量连续分布刚体绕同一轴转动 {
    int[m] {
      r(2) d m
    }
  }
}

--- 刚体的转动惯量

J = {
  圆环 {
    转动轴过中心且与圆环垂直 {
      m r(2)
    }

    转动轴沿直径 {
       m r(2) / 2
    }
  }

  薄圆盘 {
    转动轴通过中心且与圆盘垂直 {
       m r(2) / 2
    }
  }

  圆桶 {
    转动轴沿几何轴 {
       {m / 2} (r[1](2) + r[2](2))
    }
  }

  圆柱体 {
    转动轴沿几何轴 {
       m r(2) / 2
    }

    转动轴通过中心且与几何轴垂直 {
       m r(2) / 4 + m l(2) / 12
    }
  }

  细棒 {
    转动轴通过中心且与棒垂直 {
       m r(2) / 12
    }

    转动轴通过端点且与棒垂直 {
       m l(2) / 3
    }
  }

  球体 {
    转动轴沿直径 {
       2 m r(2) / 5
    }
  }

  球壳 {
    转动轴沿直径 {
     2 m r(2) / 3
    }
  }
}
