力矩的功
微分形式 {
  [
    sum[i: 1..n] { M[i] } d theta
    M d theta
  ] = d W
}

积分形式 {
  [
    int[theta[1]..theta[2]] M d theta
  ] = W
}

力矩的功率 {
  [
    d W / d t
    M {d theta / d t}
    M omega
  ] = P
}

刚体定轴转动的定能定理
转动定律 {
  [
    J alpha
    J {d omega / d t}
    J {d omega / d theta} {d theta / d t}
    J omega {d omega / d theta}
  ] = M
}

刚体定轴转动的动能定理 {
  int[theta[1]..theta[2] M d theta
  = 1/2 J omega[2](2) - 1/2 J omega[1](2)
}
