刚体定轴转动的描述

角位移  角速度  角加速度 {
  # 名词
  角位移

  # 名词
  平均角速度 {
    bar{omega} = Delta theta / Delta t
  }

  # 名词
  角速度 {
    [
      lim[Delta t to 0] Delta theta / Delta t
      d theta / d t
    ] = omega
  }

  角速度矢量 {
    # 预设
    角速度矢量的方向 {
      沿转动轴
      和刚体的旋转运动组成右手螺旋系统
    }
  }

  # 名词
  平均角加速度 {
    bar{alpha} = Delta omega / Delta t
  }


  # 名词
  角加速度 {
  [
    lim[Delta t to 0] Delta omega / Delta t
    d omega / d t
  ] = alpha
}

角量与线量的关系 {
  # 名词

  v[i] = omega r[i]

  # 名词
  加速度的切向分量
  a[i[tau]] = alpha r[i]

  # 名词
  加速度的法向分量
  a[i n] = omega(2) r[i]
